class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = None

    def insert_at_end(self, data):
        # Check if empty
        if not self.head:
            self.head = Node(data)
            return
            
        current = self.head
        while current.next:
            current = current.next
        current.next = Node(data)
